cmake_minimum_required(VERSION 3.10)
project(batt_access VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build inside Linux (the Ubuntu VM), not Windows/MinGW.")
endif()
add_library(batt_core INTERFACE)
find_package(Threads REQUIRED)
target_link_libraries(batt_core INTERFACE Threads::Threads)
target_include_directories(batt_core INTERFACE src third_party)
target_compile_options(batt_core INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
option(BATT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(BATT_THREAD_SANITIZE "Enable thread sanitizer (separate build)" OFF)
if(BATT_SANITIZE AND BATT_THREAD_SANITIZE)
  message(FATAL_ERROR "Use separate builds for thread and address sanitizers")
endif()
if(BATT_SANITIZE)
  target_compile_options(batt_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_libraries(batt_core INTERFACE -fsanitize=address,undefined)
endif()
if(BATT_THREAD_SANITIZE)
  target_compile_options(batt_core INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_libraries(batt_core INTERFACE -fsanitize=thread)
endif()
add_library(batt_sqlite STATIC third_party/sqlite/sqlite3.c)
target_include_directories(batt_sqlite PUBLIC third_party/sqlite)
target_compile_definitions(batt_sqlite PRIVATE SQLITE_THREADSAFE=1)
target_link_libraries(batt_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
if(BATT_SANITIZE)
  target_compile_options(batt_sqlite PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(BATT_THREAD_SANITIZE)
  target_compile_options(batt_sqlite PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
endif()
add_library(batt_worker src/worker.cpp src/store.cpp)
target_link_libraries(batt_worker PUBLIC batt_core batt_sqlite)
add_executable(batt-db third_party/sqlite/shell.c)
target_link_libraries(batt-db PRIVATE batt_core batt_sqlite)
target_compile_options(batt-db PRIVATE -w)
add_executable(batt-access src/access.cpp)
add_library(batt_http src/http_api.cpp)
target_include_directories(batt_http SYSTEM PRIVATE third_party/httplib)
target_link_libraries(batt_http PUBLIC batt_core)
target_link_libraries(batt-access PRIVATE batt_worker batt_http)
add_executable(batt-sim src/sim.cpp)
target_link_libraries(batt-sim PRIVATE batt_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE batt_worker)
  add_test(NAME core COMMAND core_test)
  add_executable(query_test tests/query_test.cpp)
  target_link_libraries(query_test PRIVATE batt_worker)
  add_test(NAME query_unit COMMAND query_test)
  add_executable(store_test tests/store_test.cpp)
  target_link_libraries(store_test PRIVATE batt_worker)
  add_test(NAME storage_unit COMMAND store_test)
  add_executable(api_test tests/api_test.cpp)
  target_link_libraries(api_test PRIVATE batt_worker)
  add_test(NAME api_unit COMMAND api_test)
  set_tests_properties(core query_unit storage_unit api_unit PROPERTIES TIMEOUT 15)
  find_package(PythonInterp 3 REQUIRED)
  add_test(NAME network COMMAND ${PYTHON_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py
    $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(network PROPERTIES TIMEOUT 120)
  add_test(NAME query_network COMMAND ${PYTHON_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/query_network_test.py
    $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(query_network PROPERTIES TIMEOUT 120)
  add_test(NAME persistence COMMAND ${PYTHON_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/persistence_test.py
    $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(persistence PROPERTIES TIMEOUT 120)
  add_test(NAME final_demo COMMAND ${PYTHON_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts/demo-final.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(final_demo PROPERTIES TIMEOUT 90)
  add_test(NAME http_network COMMAND ${PYTHON_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(http_network PROPERTIES TIMEOUT 120)
  add_test(NAME http_demo COMMAND ${PYTHON_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts/demo-http.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(http_demo PROPERTIES TIMEOUT 30)
  if(BATT_THREAD_SANITIZE)
    set(batt_tsan_options "$ENV{TSAN_OPTIONS}")
    if(batt_tsan_options)
      set(batt_tsan_options "${batt_tsan_options}:")
    endif()
    set_tests_properties(core query_unit storage_unit api_unit network query_network persistence final_demo http_network http_demo PROPERTIES
      ENVIRONMENT "TSAN_OPTIONS=${batt_tsan_options}halt_on_error=1:suppressions=${CMAKE_CURRENT_SOURCE_DIR}/tests/tsan.supp")
  endif()
endif()
