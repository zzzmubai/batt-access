cmake_minimum_required(VERSION 3.20)
project(batt_access VERSION 3.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build inside Linux (the Ubuntu VM), not Windows/MinGW.")
endif()
add_library(batt_core INTERFACE)
find_package(Threads REQUIRED)
target_link_libraries(batt_core INTERFACE Threads::Threads)
target_include_directories(batt_core INTERFACE src third_party)
target_compile_options(batt_core INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
option(BATT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(BATT_THREAD_SANITIZE "Enable thread sanitizer (separate build)" OFF)
if(BATT_SANITIZE AND BATT_THREAD_SANITIZE)
  message(FATAL_ERROR "Use separate builds for thread and address sanitizers")
endif()
if(BATT_SANITIZE)
  target_compile_options(batt_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_libraries(batt_core INTERFACE -fsanitize=address,undefined)
endif()
if(BATT_THREAD_SANITIZE)
  target_compile_options(batt_core INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_libraries(batt_core INTERFACE -fsanitize=thread)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(MYSQL_CONFIG mysql_config REQUIRED)
execute_process(COMMAND ${MYSQL_CONFIG} --version OUTPUT_VARIABLE MYSQL_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT MYSQL_VERSION MATCHES "^8\\.4\\.")
  message(FATAL_ERROR "Oracle MySQL 8.4 client required; found ${MYSQL_VERSION}")
endif()
execute_process(COMMAND ${MYSQL_CONFIG} --include OUTPUT_VARIABLE MYSQL_INCLUDES OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND ${MYSQL_CONFIG} --libs OUTPUT_VARIABLE MYSQL_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(MYSQL_INCLUDES UNIX_COMMAND "${MYSQL_INCLUDES}")
separate_arguments(MYSQL_LIBS UNIX_COMMAND "${MYSQL_LIBS}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS database/mysql/schema.py database/mysql/001_init.sql scripts/generate-mysql-manifest.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate-mysql-manifest.py
  ${CMAKE_CURRENT_BINARY_DIR}/generated/mysql_schema_manifest.hpp COMMAND_ERROR_IS_FATAL ANY)
add_library(batt_mysql src/db_config.cpp src/mysql_raii.cpp src/mysql_schema.cpp)
target_compile_options(batt_mysql PRIVATE ${MYSQL_INCLUDES})
target_include_directories(batt_mysql PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(batt_mysql PUBLIC batt_core ${MYSQL_LIBS})
add_library(batt_worker src/worker.cpp src/store.cpp)
target_link_libraries(batt_worker PUBLIC batt_core batt_mysql)
add_executable(batt-access src/access.cpp)
target_compile_definitions(batt-access PRIVATE BATT_VERSION="${PROJECT_VERSION}")
add_library(batt_http src/http_api.cpp)
target_include_directories(batt_http SYSTEM PRIVATE third_party/httplib)
target_link_libraries(batt_http PUBLIC batt_core)
target_link_libraries(batt-access PRIVATE batt_worker batt_http)
add_executable(batt-sim src/sim.cpp)
target_link_libraries(batt-sim PRIVATE batt_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE batt_worker)
  add_test(NAME core COMMAND core_test)
  add_executable(query_test tests/query_test.cpp)
  target_link_libraries(query_test PRIVATE batt_worker)
  add_test(NAME query_unit COMMAND query_test)
  add_executable(store_test tests/store_test.cpp)
  target_link_libraries(store_test PRIVATE batt_worker)
  add_test(NAME storage_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_mysql_cpp.py $<TARGET_FILE:store_test> --databases 2)
  add_executable(api_test tests/api_test.cpp)
  target_link_libraries(api_test PRIVATE batt_worker)
  add_test(NAME api_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_mysql_cpp.py $<TARGET_FILE:api_test>)
  set_tests_properties(core query_unit storage_unit api_unit PROPERTIES TIMEOUT 15)
  add_test(NAME network COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py
    $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(network PROPERTIES TIMEOUT 120)
  add_test(NAME query_network COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/query_network_test.py
    $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(query_network PROPERTIES TIMEOUT 120)
  add_test(NAME persistence COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/persistence_test.py
    $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(persistence PROPERTIES TIMEOUT 120)
  add_test(NAME final_demo COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts/demo-final.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(final_demo PROPERTIES TIMEOUT 90)
  add_test(NAME http_network COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py $<TARGET_FILE:batt-access> $<TARGET_FILE:batt-sim>)
  set_tests_properties(http_network PROPERTIES TIMEOUT 120)
  add_test(NAME http_demo COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts/demo-http.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(http_demo PROPERTIES TIMEOUT 30)
  add_executable(db_config_test tests/db_config_test.cpp)
  target_link_libraries(db_config_test PRIVATE batt_mysql)
  add_test(NAME database_configuration COMMAND db_config_test)
  add_executable(mysql_raii_test tests/mysql_raii_test.cpp)
  target_link_libraries(mysql_raii_test PRIVATE batt_mysql)
  add_test(NAME mysql_resources COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_mysql_cpp.py $<TARGET_FILE:mysql_raii_test>)
  add_test(NAME mysql_schema COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mysql_schema_test.py)
  set_tests_properties(database_configuration mysql_resources mysql_schema PROPERTIES TIMEOUT 30)
  if(BATT_THREAD_SANITIZE)
    set(batt_tsan_options "$ENV{TSAN_OPTIONS}")
    if(batt_tsan_options)
      set(batt_tsan_options "${batt_tsan_options}:")
    endif()
    set_tests_properties(core query_unit storage_unit api_unit network query_network persistence final_demo http_network http_demo database_configuration mysql_resources mysql_schema PROPERTIES
      ENVIRONMENT "TSAN_OPTIONS=${batt_tsan_options}halt_on_error=1")
  endif()
endif()
