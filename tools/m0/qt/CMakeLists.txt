cmake_minimum_required(VERSION 3.20)
project(batt_qt_admission LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
# Exact patch/kit is recorded after a successful build; Qt 5 is intentionally rejected.
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Test)
add_library(m0_window window.cpp window.hpp)
target_link_libraries(m0_window PUBLIC Qt6::Widgets Qt6::Network)
add_executable(qt_smoke main.cpp)
target_link_libraries(qt_smoke PRIVATE m0_window)
add_executable(qt_smoke_test smoke_test.cpp)
target_link_libraries(qt_smoke_test PRIVATE m0_window Qt6::Test)
include(CTest)
add_test(NAME qt_m0_admission COMMAND qt_smoke_test -o qt-m0-results.xml,xml -o -,txt)
set_tests_properties(qt_m0_admission PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
