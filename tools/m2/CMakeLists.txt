cmake_minimum_required(VERSION 3.20)
project(batt_mysql_resources LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build on the dedicated V3 Linux host")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_program(MYSQL_CONFIG mysql_config REQUIRED)
execute_process(COMMAND ${MYSQL_CONFIG} --include OUTPUT_VARIABLE MYSQL_INCLUDES OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND ${MYSQL_CONFIG} --libs OUTPUT_VARIABLE MYSQL_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(MYSQL_INCLUDES UNIX_COMMAND "${MYSQL_INCLUDES}")
separate_arguments(MYSQL_LIBS UNIX_COMMAND "${MYSQL_LIBS}")
find_package(Threads REQUIRED)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_library(batt_mysql ${ROOT}/src/db_config.cpp ${ROOT}/src/mysql_raii.cpp)
target_include_directories(batt_mysql PUBLIC ${ROOT}/src)
target_compile_options(batt_mysql PRIVATE ${MYSQL_INCLUDES} -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(batt_mysql PUBLIC ${MYSQL_LIBS} Threads::Threads)
add_executable(mysql_raii_test ${ROOT}/tests/mysql_raii_test.cpp)
target_link_libraries(mysql_raii_test PRIVATE batt_mysql)
add_executable(db_config_test ${ROOT}/tests/db_config_test.cpp)
target_link_libraries(db_config_test PRIVATE batt_mysql)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME database_configuration COMMAND db_config_test)
add_test(NAME mysql_resources COMMAND ${Python3_EXECUTABLE} ${ROOT}/tests/run_mysql_cpp.py $<TARGET_FILE:mysql_raii_test>)
set_tests_properties(database_configuration mysql_resources PROPERTIES TIMEOUT 30)
