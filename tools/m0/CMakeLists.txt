cmake_minimum_required(VERSION 3.20)
project(batt_v3_environment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_executable(config_test config_test.cpp)
add_test(NAME m0_config_guards COMMAND config_test)
option(BATT_M0_WITH_MYSQL "Build the MySQL 8.4 admission probe" ON)
if(BATT_M0_WITH_MYSQL)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Run the MySQL probe in the separate V3 Linux environment")
  endif()
  find_program(MYSQL_CONFIG mysql_config REQUIRED)
  execute_process(COMMAND ${MYSQL_CONFIG} --version OUTPUT_VARIABLE MYSQL_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT MYSQL_VERSION MATCHES "^8\\.4\\.")
    message(FATAL_ERROR "Oracle MySQL 8.4 C API required; found ${MYSQL_VERSION}")
  endif()
  execute_process(COMMAND ${MYSQL_CONFIG} --include OUTPUT_VARIABLE MYSQL_INCLUDES
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND ${MYSQL_CONFIG} --libs OUTPUT_VARIABLE MYSQL_LIBS
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  separate_arguments(MYSQL_INCLUDES UNIX_COMMAND "${MYSQL_INCLUDES}")
  separate_arguments(MYSQL_LIBS UNIX_COMMAND "${MYSQL_LIBS}")
  add_executable(mysql_smoke mysql_smoke.cpp)
  target_compile_options(mysql_smoke PRIVATE ${MYSQL_INCLUDES} -Wall -Wextra -Wpedantic -Wconversion)
  target_link_libraries(mysql_smoke PRIVATE ${MYSQL_LIBS})
endif()
