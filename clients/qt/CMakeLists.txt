cmake_minimum_required(VERSION 3.20)
project(batt_ops VERSION 3.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test)
add_library(batt_ops_lib api_client.cpp controller.cpp window.cpp)
target_link_libraries(batt_ops_lib PUBLIC Qt6::Widgets Qt6::Network)
target_include_directories(batt_ops_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(batt-ops main.cpp)
target_link_libraries(batt-ops PRIVATE batt_ops_lib)
include(CTest)
if(BUILD_TESTING)
  add_executable(ops_test tests.cpp)
  target_link_libraries(ops_test PRIVATE batt_ops_lib Qt6::Test)
  add_test(NAME qt_operations COMMAND ops_test -o qt-results.xml,xml -o -,txt)
  set_tests_properties(qt_operations PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS batt-ops RUNTIME DESTINATION .)
